cmake_minimum_required(VERSION 3.16)
project(topk_board_validation LANGUAGES CXX)
find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
find_package(Threads REQUIRED)
add_executable(topk_board_validation board_validation.cpp)
target_compile_features(topk_board_validation PRIVATE cxx_std_17)
target_compile_options(topk_board_validation PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(topk_board_validation PRIVATE -Wl,--no-as-needed)
target_link_libraries(topk_board_validation PRIVATE Icraft::ZG330Backend Threads::Threads)
add_library(shared_npu_board STATIC shared_npu_session.cpp)
target_compile_features(shared_npu_board PUBLIC cxx_std_17)
target_compile_options(shared_npu_board PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(shared_npu_board PUBLIC Icraft::ZG330Backend Threads::Threads)
add_executable(topk_model_board_validation model_board_validation.cpp)
target_compile_features(topk_model_board_validation PRIVATE cxx_std_17)
target_compile_options(topk_model_board_validation PRIVATE -Wall -Wextra -Wpedantic)
target_link_options(topk_model_board_validation PRIVATE -Wl,--no-as-needed)
target_link_libraries(topk_model_board_validation PRIVATE shared_npu_board)
add_executable(parameter_layout_probe parameter_layout_probe.cpp)
target_compile_features(parameter_layout_probe PRIVATE cxx_std_17)
target_compile_options(parameter_layout_probe PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(parameter_layout_probe PRIVATE Icraft::ZG330Backend)
add_executable(test_parameter_image test_parameter_image.cpp)
target_compile_features(test_parameter_image PRIVATE cxx_std_17)
target_compile_options(test_parameter_image PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME parameter_image COMMAND test_parameter_image)
add_executable(shared_binding_validation shared_binding_validation.cpp)
target_link_libraries(shared_binding_validation PRIVATE shared_npu_board)
target_compile_options(shared_binding_validation PRIVATE -Wall -Wextra -Wpedantic)
